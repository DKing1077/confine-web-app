# FULL RESET
# - wipes build cache (builder prune)
# - removes containers + networks + volumes (-v) => DATA LOSS
# - rebuilds images from scratch (no cache)
# - starts a completely fresh environment

docker builder prune -a -f
docker compose down -v
docker compose build --no-cache
docker compose up

# NORMAL REBUILD (clean rebuild)
# - removes containers + volumes (-v) => DATA LOSS
# - rebuilds images with no cache (--no-cache)
# - does NOT wipe global builder cache
# - fresh rebuild but slightly less aggressive than full reset

docker compose down -v
docker compose build --no-cache
docker compose up

# NORMAL RESTART (no rebuild, no cache clearing)
# - stops containers and removes them
# - preserves volumes (data kept)
# - starts containers again in fresh state
# - fastest option for code-only changes

docker compose down
docker compose up

# cs / js fastest

docker compose up --build
